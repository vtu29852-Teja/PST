import java.util.Arrays;
import java.util.Scanner;

public class Main {
    public static boolean isLapindrome(String s) {
        int n = s.length();
        
        // Split string into left and right halves
        String left = s.substring(0, n / 2);
        String right = s.substring((n + 1) / 2);
        
        // Convert to character arrays and sort
        char[] leftArr = left.toCharArray();
        char[] rightArr = right.toCharArray();
        
        Arrays.sort(leftArr);
        Arrays.sort(rightArr);
        
        // Compare sorted arrays
        return Arrays.equals(leftArr, rightArr);
    }

    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);
        if (sc.hasNextInt()) {
            int t = sc.nextInt();
            while (t-- > 0) {
                String s = sc.next();
                if (isLapindrome(s)) {
                    System.out.println("YES");
                } else {
                    System.out.println("NO");
                }
            }
        }
        sc.close();
    }
}
